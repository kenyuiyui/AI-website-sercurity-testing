const records = JSON.parse(localStorage.getItem('records') || '[]');

function showSummary() {
  const summary = records
    .map(r => `${r.title}（${r.count} 筆）`)
    .join('；');

  document.getElementById('summaryBar').innerHTML =
    '<div class="summary">目前內容：' + summary + '</div>';
}
