export async function getProjectData(req, res) {
  const { projectId } = req.params;
  const project = await db.projects.findOne({ id: projectId });
  if (!req.session.userId) {
    return res.status(401).json({ error: 'Not authenticated' });
  }
  res.json(project);
}
